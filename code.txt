function sendTelegramMessage(message) {
  const url = `https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendMessage`;
  const payload = { chat_id: TELEGRAM_CHAT_ID, text: message, parse_mode: 'HTML' };
  const options = { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload), muteHttpExceptions: true };
  UrlFetchApp.fetch(url, options);
}

function doPost(e) {

}

function returnJSON(data, headers) { 
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function saveBase64ToDrive(base64Str, plate, fileNamePrefix) {
  if (!base64Str) return '';
  try {
    const splitBase = base64Str.split(',');
    const type = splitBase[0].split(';')[0].replace('data:', ''); 
    const byteCharacters = Utilities.base64Decode(splitBase[1]);
    const d = new Date();
    const pad = (n) => n.toString().padStart(2, '0');
    const blob = Utilities.newBlob(byteCharacters, type, `${pad(d.getDate())}-${pad(d.getMonth()+1)}-${d.getFullYear()}-${fileNamePrefix}`);
    const rootFolder = DriveApp.getFolderById(ROOT_IMAGE_FOLDER_ID);
    const subFolderIterator = rootFolder.getFoldersByName(plate);
    const folder = subFolderIterator.hasNext() ? subFolderIterator.next() : rootFolder.createFolder(plate);
    const file = folder.createFile(blob); 
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return file.getUrl();
  } catch(e) { return 'Error saving image'; }
}

// 🟢 [ส่วนที่เพิ่มใหม่] ฟังก์ชันดึงประวัติการใช้รถทั้งหมดให้ ยพ.6
function getCarLogs() {
  try {
    const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName('CarLogs');
    if (!sheet) return { status: 'error', message: 'ไม่พบชีตชื่อ CarLogs' };
    
    const data = sheet.getDataRange().getDisplayValues();
    if (data.length < 2) return { status: 'success', data: [] }; // ถ้าชีตว่างเปล่า
    
    const headers = data[0];
    const result = [];
    // วนลูปอ่านข้อมูลตั้งแต่บรรทัดที่ 2 ลงไป (ข้าม Header)
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const obj = {};
      
      // เอาหัวคอลัมน์ภาษาไทยมาทำเป็น Key
      for (let j = 0; j < headers.length; j++) {
        const headerName = headers[j] ? headers[j].toString().trim() : "Col" + j;
        obj[headerName] = row[j] ? row[j].toString().trim() : "";
      }
      result.push(obj);
    }
    
    return { status: 'success', data: result };
  } catch(e) {
    return { status: 'error', message: e.toString() };
  }
}
